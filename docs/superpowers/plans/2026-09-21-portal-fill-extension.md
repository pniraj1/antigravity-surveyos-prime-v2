# Portal Fill Extension v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One click in SurveyOS's NIA dialog hands a finished claim to a
Chrome extension, which fills the New India non-tieup survey page and
attaches the assessment Excel — never submitting, never touching credentials.

**Architecture:** SurveyOS builds a portal-neutral `PortalPacket` (canonical
field names + the Excel as base64) and posts it to the page; the extension's
Relay content script stores it. On the portal, a Panel content script shows
the packet and a main-world Filler writes each field through Angular's own
`ngModelController.$setViewValue` so `ng-change` and validators run as if
typed. A generated field map (canonical name → `data-ng-model` path) lives in
`extension/maps/` and is also published as static JSON on our hosting.

**Tech Stack:** SurveyOS side — TypeScript, React 19, Vitest 4, ExcelJS.
Extension — Manifest V3, plain JavaScript with `// @ts-check` JSDoc (no
bundler: content scripts are classic scripts, shared libs use a two-line UMD
shim so Vitest can import them). Map generator — Python 3 stdlib.

**Spec:** `docs/superpowers/specs/2026-09-20-portal-fill-extension-design.md`

`C:\Users\Manasi\Downloads\The New India Assurance Co. Ltd MAP_ worklistnew.html`

## Global Constraints

- E1 — no login-page permission, no reading password fields, nothing stored but packet + map.
- E2 — the extension never dispatches a click on Validate / Submit / Save / upload-confirm.
- E3 — claim numbers compared after `normaliseClaimNo` (uppercase, `[^A-Z0-9]` stripped); refusal shows both.
- E4 — a value counts as "existing" only when it differs from the map's `empty` for that field.
- E5 — only network call: `GET https://motorsurveyos-in.web.app/portal-maps/nia-nontieup.json`.
- E6 — packet cleared after a successful fill or 24 h.
- E7 — no AI anywhere in the extension.
- E8 — per-field failure; `notFound` / `manual` / `kept` / `filled` per field.
- E9 — `extension/maps/nia-nontieup.json` is generated by `scripts/portal-map/build-ext-map.py`; never hand-edited.
- E10 — writes go through `ngModelController.$setViewValue(v)` + `$render()` inside `scope.$evalAsync`; never a bare scope assignment.
- Portal attributes are `data-ng-model`; selectors match `[ng-model="p"], [data-ng-model="p"]`.
- Two corrections to the spec's field table, found in the capture: `provisionAmount`

```ts
// src/lib/portal/packet.ts
/**
 * Portal-neutral hand-off from a finished claim to the portal-fill extension.
 * Canonical names are the contract; the extension's field map turns them into
 * portal model paths. Nothing here knows an Angular path.
 */
import type { ClaimData } from '@/types/claim';
import { getCompulsoryExcess } from '@/lib/calculations/assessment';
import { calculateFeeSummary } from '@/lib/calculations/fees';
import { toDepreciationType } from '@/lib/calculations/depreciation';
import {
  buildNiaWorkbook,
  niaAssessmentRows,
  niaPainting,
} from '@/lib/reports/nia-assessment-excel';

export const CANONICAL_FIELDS = [
  'surveyDate', 'surveyTime', 'surveyPlace',
  'claimantMobile', 'claimantEmail',
  'vendorInvoiceNo', 'vendorInvoiceDate', 'gstApplicable',
  'paintingEstimate', 'paintingAssessed',
  'noOfInvoices', 'netSalvage', 'compulsoryExcess', 'voluntaryExcess',
  'feeInvoiceDate', 'professionalFee', 'photosAmount', 'conveyanceAmount',
  'feeOthers', 'feeGstApplicable', 'feeGstPercent',
] as const;
export type CanonicalField = (typeof CANONICAL_FIELDS)[number];

export interface PortalPacket {
  v: 1;
  portal: 'nia';
  flow: 'nontieup';
  claimNo: string;
  builtAt: string;
  fields: Partial<Record<CanonicalField, string | number>>;
  files: Array<{ slot: 'assessmentExcel'; name: string; mime: string; base64: string }>;
}

const XLSX_MIME = 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet';

function bytesToBase64(buf: ArrayBuffer): string {
  const bytes = new Uint8Array(buf);
  let s = '';
  for (let i = 0; i < bytes.length; i += 0x8000) {
    s += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  }
  return btoa(s);
}

/** Drop empty strings, null and undefined so the extension never "fills" a blank. */
function compact(o: Record<string, string | number | null | undefined>) {
  const out: Partial<Record<CanonicalField, string | number>> = {};
  for (const [k, v] of Object.entries(o)) {
    if (v === '' || v === null || v === undefined) continue;
    out[k as CanonicalField] = v;
  }
  return out;
}

export async function buildPortalPacket(
  claim: ClaimData,
  opts: { ageMonths: number },
): Promise<PortalPacket> {
  const claimNo = (claim.policy?.claimNumber ?? '').trim();
  if (!claimNo) throw new Error('This claim has no claim number; the portal needs one to match on.');

  const depType = toDepreciationType(claim.depreciationType);
  const rows = claim.assessmentRows ?? [];
  const fee = claim.feeBill;
  const feeSummary = calculateFeeSummary(fee);
  const painting = niaPainting(rows);
  const isNilDep = depType === 'nil';

  const sheetRows = niaAssessmentRows(rows, opts.ageMonths, depType);
  const wb = buildNiaWorkbook(sheetRows);
  const buf = await wb.xlsx.writeBuffer();
  const safeNo = claimNo.replace(/[^A-Za-z0-9_-]+/g, '_');

  return {
    v: 1,
    portal: 'nia',
    flow: 'nontieup',
    claimNo,
    builtAt: new Date().toISOString(),
    fields: compact({
      surveyDate: claim.accident?.dateOfSurvey,
      surveyTime: claim.accident?.timeOfSurvey,
      surveyPlace: claim.accident?.placeOfSurvey,
      claimantMobile: claim.policy?.insuredMobile,
      vendorInvoiceNo: claim.billCheck?.billNo,
      vendorInvoiceDate: claim.billCheck?.billDate,
      gstApplicable: rows.some((r) => (r.gst ?? 0) > 0) ? 'Y' : 'N',
      // R11: on nil-dep the painting is a labour row inside the file; the
      // Painting section must stay empty.
      paintingEstimate: isNilDep ? undefined : painting.estimate,
      paintingAssessed: isNilDep ? undefined : painting.assessed,
      noOfInvoices: '1',
      netSalvage: fee?.billSalvage ?? fee?.salvageValue ?? 0,
      compulsoryExcess: getCompulsoryExcess(fee),
      voluntaryExcess: fee?.voluntaryExcess ?? 0,
      feeInvoiceDate: fee?.billDate,
      professionalFee: fee?.professionalFee ?? 0,
      photosAmount: feeSummary.photographyCharges,
      conveyanceAmount: feeSummary.travelExpenses,
      feeOthers: (fee?.postalCharges ?? 0) + (fee?.haltageCharges ?? 0),
      feeGstApplicable: fee?.includeGST ? 'Y' : 'N',
      feeGstPercent: fee?.includeGST ? 18 : undefined,
    }),
    files: [
      {
        slot: 'assessmentExcel',
        name: `${safeNo}_Primary_Assessment_Template.xlsx`,
        mime: XLSX_MIME,
        base64: bytesToBase64(buf as ArrayBuffer),
      },
    ],
  };
}
```

```python
# scripts/portal-map/build-ext-map.py
"""
Overlay + extractor JSON -> extension map + hosted copy + PII-free fixture.

  python scripts/portal-map/build-ext-map.py nia-nontieup

Fails loudly when an overlay model path is not present in the capture (E9).
"""
import json, sys, html, pathlib

ROOT = pathlib.Path(__file__).resolve().parents[2]
HERE = pathlib.Path(__file__).resolve().parent

def load(p):
    return json.loads(p.read_text(encoding="utf8"))

def main(flow):
    overlay = load(HERE / f"{flow}.overlay.json")
    capture = load(HERE / f"{flow}-survey.json")
    by_model = {}
    for fields in capture["groups"].values():
        for f in fields:
            if f.get("model"):
                by_model.setdefault(f["model"], []).append(f)

    missing = [k for k, v in {**overlay["fields"], **overlay["files"]}.items()
               if v["model"] not in by_model]
    if missing:
        sys.exit(f"overlay paths not in capture: {missing}")

    def entry(v):
        e = {"modelPath": v["model"], "kind": v.get("kind", "text"), "empty": v.get("empty", "")}
        for k in ("transform", "options", "maxLen", "from", "constant", "onlyIf"):
            if k in v:
                e[k] = v[k]
        return e

    out = {k: overlay[k] for k in ("portal", "flow", "version", "route", "controller",
                                    "claimNoPath", "claimNoFallback", "invoiceCountPath")}
    out["fields"] = {k: entry(v) for k, v in overlay["fields"].items()}
    out["files"] = {k: {"modelPath": v["model"]} for k, v in overlay["files"].items()}

    text = json.dumps(out, indent=2) + "\n"
    for dest in (ROOT / "extension/maps" / f"{flow}.json", ROOT / "public/portal-maps" / f"{flow}.json"):
        dest.parent.mkdir(parents=True, exist_ok=True)
        dest.write_text(text, encoding="utf8")

    write_fixture(flow, overlay, by_model)
    print("ok", len(out["fields"]), "fields")

def write_fixture(flow, overlay, by_model):
    """One input per overlay path, same data-ng-model / type / name / radio values as
    the capture, no captured values. AngularJS 1.4.3 from cdnjs; a stub controller
    that mirrors the model roots so $setViewValue has somewhere to land."""
    rows = []
    for canon, v in {**overlay["fields"], **overlay["files"]}.items():
        for f in by_model[v["model"]]:
            t = f.get("type", "text")
            name = html.escape(f.get("name") or canon)
            model = html.escape(f["model"])
```

```js
// extension/src/lib/packet-validate.js
// @ts-check
/** Shape check for a PortalPacket coming over window.postMessage. Data, never code. */
const NS = 'SxPacket';
const MAX_FILE_B64 = 1_400_000; // ≈1 MB of bytes
const SLOTS = ['assessmentExcel'];

/** @param {unknown} x */
function validatePacket(x) {
  const fail = (/** @type {string} */ reason) => ({ ok: /** @type {const} */ (false), reason });
  if (!x || typeof x !== 'object') return fail('not an object');
  const p = /** @type {Record<string, any>} */ (x);
  if (p.v !== 1) return fail('unsupported packet version');
  if (p.portal !== 'nia' || p.flow !== 'nontieup') return fail('packet is for another portal');
  if (typeof p.claimNo !== 'string' || !p.claimNo.trim()) return fail('missing claim number');
  if (typeof p.builtAt !== 'string') return fail('missing builtAt');
  if (!p.fields || typeof p.fields !== 'object') return fail('missing fields');
  for (const [k, v] of Object.entries(p.fields)) {
    if (!(typeof v === 'string' || typeof v === 'number')) return fail(`field ${k} is not a scalar`);
  }
  if (!Array.isArray(p.files)) return fail('missing files');
  for (const f of p.files) {
    if (!f || !SLOTS.includes(f.slot)) return fail('unknown file slot');
    if (typeof f.name !== 'string' || typeof f.mime !== 'string') return fail('bad file meta');
    if (typeof f.base64 !== 'string' || !/^[A-Za-z0-9+/]*={0,2}$/.test(f.base64)) return fail('bad base64');
    if (f.base64.length > MAX_FILE_B64) return fail('file too large');
  }
  return { ok: /** @type {const} */ (true), packet: p };
}

const API = { validatePacket };
// ---- umd shim: classic script in the extension, ESM/CJS in Vitest ----
if (typeof module !== 'undefined' && module.exports) module.exports = API;
else globalThis[NS] = API;
```

```js
// Runs on motorsurveyos-in.web.app. Marks the page so SurveyOS can show the
// "Send to extension" button, and moves a packet into chrome.storage.local.
(() => {
  const { validatePacket } = globalThis.SxPacket;
  const VERSION = chrome.runtime.getManifest().version;
  document.documentElement.dataset.surveyosExt = VERSION;

  window.addEventListener('message', (ev) => {
    if (ev.source !== window || ev.origin !== location.origin) return;
    const d = ev.data;
    if (!d || d.type !== 'surveyos:portal-packet') return;
    const r = validatePacket(d.packet);
    if (!r.ok) {
      window.postMessage({ type: 'surveyos:portal-packet-ack', ok: false, reason: r.reason }, location.origin);
      return;
    }

      const err = chrome.runtime.lastError;
      window.postMessage(
        { type: 'surveyos:portal-packet-ack', ok: !err, reason: err ? err.message : undefined },
        location.origin,
      );
    });
  });
})();
```

```js
// extension/src/lib/fill-core.js
// @ts-check
/**
 * Pure fill logic. No DOM, no Angular — the Filler feeds it readers and
 * executes the steps it returns. Everything testable lives here.
 */
const NS = 'SxFill';

/** @param {string} s */
const normaliseClaimNo = (s) => String(s ?? '').toUpperCase().replace(/[^A-Z0-9]/g, '');

const pad2 = (/** @type {string|number} */ n) => String(n).padStart(2, '0');

const transforms = {
  /** ISO date / ISO datetime / dd/MM/yyyy -> dd/MM/yyyy */
  ddmmyyyy: (/** @type {unknown} */ v) => {
    const s = String(v ?? '').trim().split('T')[0];
    let m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
    if (m) return `${pad2(m[3])}/${pad2(m[2])}/${m[1]}`;
    m = s.match(/^(\d{1,2})[/.-](\d{1,2})[/.-](\d{4})$/);
    if (m) return `${pad2(m[1])}/${pad2(m[2])}/${m[3]}`;
    return s;
  },
  /** 9:05 / 14:30:00 -> HH:MM */
  hhmm: (/** @type {unknown} */ v) => {
    const m = String(v ?? '').trim().match(/^(\d{1,2}):(\d{2})/);
    return m ? `${pad2(m[1])}:${m[2]}` : String(v ?? '');
  },
  money2dp: (/** @type {unknown} */ v) => (Number(v) || 0).toFixed(2),
  yn: (/** @type {unknown} */ v) => (v === true || v === 'Y' || v === 'y' || v === 'yes' ? 'Y' : 'N'),
};

/**
 * E4: "empty" = the portal's own default for the field. Numbers and numeric
 * strings equal to 0 are empty on money fields; null/undefined/'' always are.
 * @param {unknown} current @param {unknown} empty
 */
function isEmptyValue(current, empty) {
  if (current === null || current === undefined || current === '') return true;
  if (empty !== null && empty !== undefined && current === empty) return true;
  if (empty === '' && (current === 0 || current === '0' || current === '0.00')) return true;
  return false;
}

/** @param {unknown} current @param {unknown} empty @param {string} incoming */
function decide(current, empty, incoming) {
  if (isEmptyValue(current, empty)) return 'fill';
  const cur = current instanceof Date ? null : String(current);
  if (cur !== null && (cur === incoming || (Number(cur) === Number(incoming) && !Number.isNaN(Number(cur))))) return 'same';
  return 'kept';
}

/**
 * @param {any} map @param {any} packet
 * @param {{ current: (p: string) => unknown, radioValues: (p: string) => string[] }} read
 */
function planFill(map, packet, read) {
  const fields = packet.fields ?? {};
  /** @type {any[]} */
  const steps = [];
  for (const [name, f] of Object.entries(map.fields)) {
    const spec = /** @type {any} */ (f);
    const base = { name, modelPath: spec.modelPath, kind: spec.kind };

    /** @type {string|undefined} */
    let value;
    if (spec.kind === 'dateRadioOrOthers') {
      const src = fields[spec.from];
      if (src === undefined) { steps.push({ ...base, action: 'skip', why: 'no source' }); continue; }
      const dmy = transforms.ddmmyyyy(src);
      value = read.radioValues(spec.modelPath).includes(dmy) ? dmy : 'Others';
    } else if ('constant' in spec) {
      if (spec.onlyIf && fields[spec.onlyIf] === undefined) { steps.push({ ...base, action: 'skip', why: 'condition not met' }); continue; }
      value = String(spec.constant);
    } else {
      const raw = fields[name];
      if (raw === undefined) { steps.push({ ...base, action: 'skip', why: 'not in packet' }); continue; }
      value = spec.kind === 'money' ? transforms.money2dp(raw)
        : spec.transform ? transforms[spec.transform](raw)
        : String(raw);
      if (spec.options) value = spec.options[value] ?? value;
      if (spec.maxLen) value = value.slice(0, spec.maxLen);
    }

    const current = read.current(spec.modelPath);
    const action = decide(current, spec.empty, value);
    steps.push({ ...base, action, value, current });
  }
  return steps;
}

const API = { normaliseClaimNo, transforms, isEmptyValue, decide, planFill };
// ---- umd shim: classic script in the extension, ESM/CJS in Vitest ----
if (typeof module !== 'undefined' && module.exports) module.exports = API;
else globalThis[NS] = API;
```

```js
(() => {
  const { normaliseClaimNo, planFill } = globalThis.SxFill;
  const ng = () => /** @type {any} */ (globalThis).angular;

  const q = (/** @type {string} */ p) =>
    /** @type {HTMLElement[]} */ (Array.from(document.querySelectorAll(`[ng-model="${p}"], [data-ng-model="${p}"]`)));

  function ctrlScope(/** @type {any} */ map) {
    const el = document.querySelector(`[ng-controller="${map.controller}"], [data-ng-controller="${map.controller}"]`);
    return el ? ng().element(el).scope() : null;
  }

  function $parse() { return ng().element(document.body).injector().get('$parse'); }

  function readPath(/** @type {any} */ scope, /** @type {string} */ path) {
    if (path.startsWith('sessionStorage:')) return sessionStorage.getItem(path.slice('sessionStorage:'.length));
    try { return $parse()(path)(scope); } catch { return undefined; }
  }

  function probe(/** @type {any} */ map) {
    if (!ng()) return { ok: false, reason: 'AngularJS not found on this page' };
    const scope = ctrlScope(map);
    if (!scope) return { ok: false, reason: `controller ${map.controller} not found — portal changed?` };
    const claimNo = readPath(scope, map.claimNoPath) ?? readPath(scope, map.claimNoFallback);
    const invoices = readPath(scope, map.invoiceCountPath);
    return { ok: true, claimNo: claimNo ? String(claimNo) : '', invoices: Array.isArray(invoices) ? invoices.length : 1 };
  }

  /** E10: write like a keystroke. Returns 'filled' | 'manual'. */
  function write(/** @type {HTMLElement} */ el, /** @type {string} */ value, /** @type {string} */ kind) {
    const ctrl = ng().element(el).controller('ngModel');
    if (!ctrl) return 'manual';
    const scope = ng().element(el).scope();
    scope.$evalAsync(() => {
      ctrl.$setViewValue(value);
      ctrl.$render();
      if (kind !== 'select') el.dispatchEvent(new Event('input', { bubbles: true }));
    });
    return 'filled';
  }

  function attachFile(/** @type {HTMLElement} */ el, /** @type {{name:string,mime:string,base64:string}} */ f) {
    const bin = atob(f.base64);
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    const dt = new DataTransfer();
    dt.items.add(new File([bytes], f.name, { type: f.mime }));
    /** @type {HTMLInputElement} */ (el).files = dt.files;
    el.dispatchEvent(new Event('change', { bubbles: true }));
  }

  function fill(/** @type {any} */ map, /** @type {any} */ packet) {
    const p = probe(map);
    if (!p.ok) return { ok: false, reason: p.reason, steps: [], files: [] };
    if (normaliseClaimNo(p.claimNo) !== normaliseClaimNo(packet.claimNo)) {
      return { ok: false, reason: `claim mismatch — page: ${p.claimNo || '(none)'}, packet: ${packet.claimNo}`, steps: [], files: [] };
    }
    const scope = ctrlScope(map);
    const read = {
      current: (/** @type {string} */ path) => {
        const el = q(path)[0];
        const c = el ? ng().element(el).controller('ngModel') : null;
        return c ? c.$modelValue : readPath(scope, path);
      },
      radioValues: (/** @type {string} */ path) => q(path).map((el) => /** @type {HTMLInputElement} */ (el).value),
    };
    const steps = planFill(map, packet, read).map((s) => {
      if (s.action !== 'fill') return s;
      const els = q(s.modelPath);
      if (!els.length) return { ...s, action: 'notFound' };
      const el = s.kind === 'radio' || s.kind === 'dateRadioOrOthers'
        ? els.find((e) => /** @type {HTMLInputElement} */ (e).value === s.value)
        : els[0];
      if (!el) return { ...s, action: 'notFound', why: 'no option with that value' };
      if (/** @type {HTMLInputElement} */ (el).disabled) return { ...s, action: 'manual', why: 'disabled on the page' };
      try {
        const r = write(el, /** @type {string} */ (s.value), s.kind);
        if (r === 'filled') { el.style.outline = '2px solid #16a34a'; return { ...s, action: 'filled' }; }
        return { ...s, action: 'manual', why: 'not Angular-bound' };
      } catch (e) {
        return { ...s, action: 'error', why: String(e) };
      }
    });
    for (const s of steps) if (s.action === 'kept') { const el = q(s.modelPath)[0]; if (el) el.style.outline = '2px solid #f59e0b'; }

    const files = (packet.files ?? []).map((/** @type {any} */ f) => {
      const spec = map.files?.[f.slot];
      const el = spec ? q(spec.modelPath)[0] : null;
      if (!el) return { slot: f.slot, status: 'notFound' };
      try { attachFile(el, f); return { slot: f.slot, status: 'attached' }; }
      catch (e) { return { slot: f.slot, status: 'error', why: String(e) }; }
    });
    return { ok: true, steps, files };
  }

  window.addEventListener('message', (ev) => {
    if (ev.source !== window) return;
    const d = ev.data;
    if (!d || !d.sx || !d.type) return;
    if (d.type === 'probe') window.postMessage({ sx: d.sx, type: 'probe:result', ...probe(d.map) }, '*');
    if (d.type === 'fill') window.postMessage({ sx: d.sx, type: 'fill:result', ...fill(d.map, d.packet) }, '*');
  });
})();
```

```js
const MAP_URL = 'https://motorsurveyos-in.web.app/portal-maps/nia-nontieup.json';
const DAY_MS = 24 * 60 * 60 * 1000;

async function loadBundledMap() {
  const r = await fetch(chrome.runtime.getURL('maps/nia-nontieup.json'));
  return r.json();
}

async function refreshMap() {
  const { map: current } = await chrome.storage.local.get('map');
  const bundled = await loadBundledMap();
  let best = current && current.version >= bundled.version ? current : bundled;
  try {
    const r = await fetch(MAP_URL, { cache: 'no-store' });
    if (r.ok) {
      const remote = await r.json();
      if (remote && remote.portal === 'nia' && remote.flow === 'nontieup' && remote.version > best.version) best = remote;
    }
  } catch { /* offline: keep what we have */ }
  await chrome.storage.local.set({ map: best });
}

async function expirePacket() {
  const { storedAt } = await chrome.storage.local.get('storedAt');
  if (storedAt && Date.now() - storedAt > DAY_MS) await chrome.storage.local.remove(['packet', 'storedAt']);
}

chrome.runtime.onInstalled.addListener(() => {
  refreshMap();
  chrome.alarms.create('sx-daily', { periodInMinutes: 24 * 60 });
});
chrome.runtime.onStartup.addListener(refreshMap);
chrome.alarms.onAlarm.addListener((a) => { if (a.name === 'sx-daily') { refreshMap(); expirePacket(); } });
chrome.storage.onChanged.addListener((ch, area) => { if (area === 'local' && ch.packet) expirePacket(); });
```

```ts
// src/lib/portal/extension-bridge.ts
import type { PortalPacket } from './packet';

/** The Relay content script stamps the page when the extension is installed. */
export function isExtensionPresent(): boolean {
  return typeof document !== 'undefined' && !!document.documentElement.dataset.surveyosExt;
}

export function sendPacketToExtension(
  packet: PortalPacket,
  timeoutMs = 3000,
): Promise<{ ok: true } | { ok: false; reason: string }> {
  return new Promise((resolve) => {
    const onAck = (ev: MessageEvent) => {
      if (ev.origin !== location.origin || ev.source !== window) return;
      if (!ev.data || ev.data.type !== 'surveyos:portal-packet-ack') return;
      cleanup();
      resolve(ev.data.ok ? { ok: true } : { ok: false, reason: ev.data.reason ?? 'extension refused the packet' });
    };
    const timer = setTimeout(() => { cleanup(); resolve({ ok: false, reason: 'No reply from the extension. Is it enabled?' }); }, timeoutMs);
    const cleanup = () => { clearTimeout(timer); window.removeEventListener('message', onAck); };
    window.addEventListener('message', onAck);
    window.postMessage({ type: 'surveyos:portal-packet', packet }, location.origin);
  });
}
```

```python
dist = EXT / "dist"; dist.mkdir(exist_ok=True)
```
